=== script 1: index check (run once per enriched index)
"""Check an enriched index against the enrichment file that built it. Read-only.

1. document count (Lucene) == 6,044, same id set as the base index;
2. for EVERY document: stored expansion == read_expansions() of today's code;
3. sampled accepted terms (seed 42, 2 documents per catalogue): every analysed
   token of every accepted term is a posting of that document's expansion field;
4. documents whose expansion is empty, listed;
5. manifest provenance: file, SHA-256 (recomputed), model, service, run.
"""
import json
import random
import sys
from pathlib import Path

sys.path.insert(0, "src")
from pyserini.index.lucene import LuceneIndexReader  # noqa: E402
from sira_cti.common import read_jsonl  # noqa: E402
from sira_cti.index.build_enriched import EXPANSION_FIELD, file_sha256, read_expansions  # noqa: E402

index_dir = Path(sys.argv[1])
manifest = json.loads((index_dir / "manifest.json").read_text())
enrichment = Path(manifest["enrichment_path"])
print(f"== {index_dir}")
ok = True

# 5. provenance
sha = file_sha256(enrichment)
print(f"  built from        {enrichment}")
recorded = manifest.get("enrichment_sha256")
print(f"  sha256            {recorded or 'not recorded (built before the field existed)'}  "
      f"{'recomputed matches: ' + str(recorded == sha) if recorded else 'file now: ' + sha}")
print(f"  model / service   {manifest['enrichment_model']} / {manifest.get('enrichment_service')} (api {manifest.get('enrichment_api')})  run {manifest.get('enrichment_run')}")
print(f"  kb_dir / config   {manifest['kb_dir']} / {manifest['config_hash']}")
if manifest.get("enrichment_sha256") is not None and manifest["enrichment_sha256"] != sha:
    ok = False

# 1. document count and id set
reader = LuceneIndexReader(str(index_dir))
base = LuceneIndexReader("indexes/base")
n, nb = reader.stats()["documents"], base.stats()["documents"]
ids = [reader.convert_internal_docid_to_collection_docid(i) for i in range(n)]
base_ids = {base.convert_internal_docid_to_collection_docid(i) for i in range(nb)}
same_set = set(ids) == base_ids
print(f"  documents         {n} (base {nb}); same id set as base: {same_set}; manifest doc_count {manifest['doc_count']}")
ok &= n == 6044 and same_set and manifest["doc_count"] == 6044

# 2. every stored expansion == today's code
expected = read_expansions(enrichment, index_repaired_ids=manifest.get("index_repaired_ids", False))
stored = {d: json.loads(reader.doc(d).raw()).get(EXPANSION_FIELD, "") for d in ids}
mismatch = [d for d in ids if stored[d] != expected.get(d, "")]
print(f"  stored expansion == read_expansions() today: {n - len(mismatch)} of {n}" + (f"  MISMATCH {mismatch[:10]}" if mismatch else ""))
ok &= not mismatch

# 4. empty expansions
empty = sorted(d for d in ids if not stored[d].strip())
print(f"  empty expansion   {len(empty)} documents: {empty}")
print(f"  manifest          documents_with_expansion {manifest['documents_with_expansion']} (= {n} - {len(empty)}: "
      f"{manifest['documents_with_expansion'] == n - len(empty)}), accepted_terms {manifest['accepted_terms']}, "
      f"parse_failures {manifest['parse_failures']}")
ok &= manifest["documents_with_expansion"] == n - len(empty)

# 3. postings for sampled accepted terms
recs = {r.doc_id: r for r in read_jsonl(enrichment)}
rng = random.Random(42)
sample = []
for src in ("cwe", "capec", "attack", "cve"):
    pool = sorted(d for d, r in recs.items() if r.source.value == src and r.accepted_terms)
    sample += rng.sample(pool, 2)
# pyserini's get_postings_list reads "contents" only; go to Lucene for the expansion field.
from jnius import autoclass  # noqa: E402
MultiTerms = autoclass("org.apache.lucene.index.MultiTerms")
Term = autoclass("org.apache.lucene.index.Term")


def posting_in_expansion(token: str, doc_id: str) -> bool:
    target = reader.convert_collection_docid_to_internal_docid(doc_id)
    pe = MultiTerms.getTermPostingsEnum(reader.reader, EXPANSION_FIELD, Term(EXPANSION_FIELD, token).bytes())
    return pe is not None and pe.advance(target) == target


# Negative control: a token that is in the document's expansion must NOT be found
# under a document whose expansion lacks it (so the lookup can say no).
terms = tokens = found = 0
misses = []
for d in sample:
    for t in recs[d].accepted_terms:
        # The text actually indexed for a structural term is its normalised id (expansion_query()).
        text = t.structural_id if t.structural_id else t.term
        terms += 1
        for tok in reader.analyze(text):
            tokens += 1
            if posting_in_expansion(tok, d):
                found += 1
            else:
                misses.append((d, text, tok))
control_tok = reader.analyze(recs[sample[0]].accepted_terms[0].term)[0]
# a document WITH an expansion, whose analysed expansion lacks the token
control_doc = next(d for d in ids if stored[d].strip() and control_tok not in reader.analyze(stored[d]))
control_ok = not posting_in_expansion(control_tok, control_doc)
print(f"  negative control  {control_tok!r} under {control_doc} (non-empty expansion without it): "
      f"found={not control_ok} (must be False)")
ok &= control_ok
print(f"  postings          {terms} accepted terms from {len(sample)} sampled documents {sample}")
print(f"                    {found} of {tokens} analysed tokens are postings of the right document's expansion field"
      + (f"  MISSING {misses[:10]}" if misses else ""))
ok &= terms >= 20 and not misses
print(f"  RESULT            {'PASS' if ok else 'FAIL'}")

    except FileNotFoundError:

def title(r, doc_id):
    text = r.doc_text(doc_id)
    return (text.split(" {")[0] if text else doc_id)[:38]

    qtok = set(R["base"].analyze(q))
    for name, r in R.items():
        hits = r.retrieve(q, k=5).hits
        tops[(q, name)] = [h.doc_id for h in hits]
        print(f"  {name}")
        for i, h in enumerate(hits, 1):
            via = ""
