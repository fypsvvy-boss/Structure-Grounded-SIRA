=== the script (scratch; matches every figure in vertex_sample_vs_aistudio_gemini.txt) ===
"""Compare Gemini enrichment files: structural ids, rejections, counting runs, overlap, tokens.
usage: compare_runs.py [--ids A,B] label=path price_in price_out ... (prices via --price label=in,out)"""
import sys, itertools, statistics
sys.path.insert(0, "src")
from sira_cti.common import read_jsonl, RejectReason, TermKind

args = sys.argv[1:]
only = None; prices = {}
runs = []
for a in args:
    if a.startswith("--ids="): only = set(a[6:].split(","))
    elif a.startswith("--price="):
        lab, v = a[8:].split("="); pi, po = map(float, v.split(",")); prices[lab] = (pi, po)
    else:
        lab, p = a.split("=", 1); runs.append((lab, p))

data = {}
for lab, p in runs:
    recs = {r.doc_id: r for r in read_jsonl(p) if only is None or r.doc_id in only}
    data[lab] = recs
docs = sorted(set.intersection(*(set(d) for d in data.values())))
print(f"documents in all files: {len(docs)}")
for lab in data:
    recs = [data[lab][d] for d in docs]
    st = [t for r in recs for t in r.proposed_terms if t.kind is TermKind.STRUCTURAL]
    acc = [t for t in st if t.accepted]
    by = {}
    for t in st:
        if not t.accepted:
            by[t.reject_reason.value] = by.get(t.reject_reason.value, 0) + 1
    exist = sum(by.get(k, 0) for k in ("not_in_graph", "deprecated", "revoked", "malformed_id"))
    counting = sum(1 for t in st if getattr(t, "in_counting_run", False))
    n_props = [len(r.proposed_terms) for r in recs]
    unparse = sum(1 for r in recs if any(t.reject_reason is RejectReason.LLM_JSON_ERROR for t in r.proposed_terms))
    tok = [r.tokens for r in recs]
    P = sum(t.prompt for t in tok); C = sum(t.completion for t in tok); T = sum(t.thinking for t in tok)
    line = (f"{lab:10s} structural proposed {len(st)}, accepted {len(acc)}, rejected {by or '{}'}; "
            f"existence {exist}, name {by.get('name_mismatch',0)}, counting-run ids {counting}; "
            f"proposals/reply min {min(n_props)} median {statistics.median(n_props)} max {max(n_props)}; unparseable {unparse}")
    print(line)
    calls = sum(r.llm_calls for r in recs)
    s = f"           tokens: prompt {P} ({P/len(recs):.0f}/doc), reply {C} ({C/len(recs):.0f}/doc), thinking {T} ({T/len(recs):.0f}/doc), calls {calls}"
    if lab in prices:
        pi, po = prices[lab]; cost = (P*pi + (C+T)*po)/1e6
        s += f"; cost ${cost:.4f} (${cost/len(recs):.5f}/doc at ${pi}/${po})"
    print(s)
    rejected_list = [(r.doc_id, t.structural_id or t.term, t.reject_reason.value) for r in recs for t in r.proposed_terms
                     if t.kind is TermKind.STRUCTURAL and not t.accepted]
    print(f"           rejected: {rejected_list}")

def accepted(lab):
    return {(d, t.structural_id) for d in docs for t in data[lab][d].proposed_terms
            if t.kind is TermKind.STRUCTURAL and t.accepted and t.structural_id}
def props(lab, d):
    return [(t.term, t.kind.value, t.claimed_name) for t in data[lab][d].proposed_terms]
print("\naccepted structural ids shared (document, id), shared / union:")
for a, b in itertools.combinations(data, 2):
    A, B = accepted(a), accepted(b)
    same = sum(1 for d in docs if props(a, d) == props(b, d))
    print(f"  {a} vs {b}: {len(A & B)} of {len(A | B)} = {len(A & B)/max(1,len(A|B)):.0%}   replies identical {same} of {len(docs)}")
