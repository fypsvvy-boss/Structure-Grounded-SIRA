def struct_ids(run, d):
    return {(d, t.structural_id) for t in run[d].proposed_terms
            if t.kind is TermKind.STRUCTURAL and t.accepted and t.structural_id}


def free(run, d, f):
    return {(d, f(t.term)) for t in run[d].proposed_terms if t.kind is not TermKind.STRUCTURAL and t.accepted}


def overlap(a, b, fn):
    A = set().union(*(fn(a, d) for d in docs))
    B = set().union(*(fn(b, d) for d in docs))
    return len(A & B), len(A | B)


def ident(a, b, subset):
    return sum(1 for d in subset if proposals(a[d]) == proposals(b[d]))

                                 capec_path=cfg["capec_path"], domains=cfg.get("domains"))


def family(a, b):  # ATT&CK parent/child or siblings: T1053 / T1053.005 / T1053.001
    return a.startswith("T") and b.startswith("T") and a.split(".")[0] == b.split(".")[0]


def rel(run, *, by_source=None, drop_family=False):
    hops, unreach, no_anchor = [], 0, 0
    for d in docs:
        rec = run[d]
        if by_source and rec.source.value != by_source:
            continue
        anchor = graph.resolve(d)
        shown = None
        for t in rec.structural_terms:
            if not t.accepted:
                continue
            if shown is None:
                shown = record_shown_text(rec)
            if structural_provenance(t.structural_id, d, shown) != "not_found":
                continue
            if drop_family and family(d, t.structural_id):
                continue
            if anchor is None:
                no_anchor += 1
            elif t.graph_distance is None:
                unreach += 1
            else:
                hops.append(t.graph_distance)
    return hops, unreach, no_anchor


def line(label, hops, unreach, no_anchor):
    m = len(hops)
    a = m + unreach
    med = statistics.median(hops) if hops else "-"
    med7 = statistics.median(hops + [7] * unreach) if a else "-"
    print(f"  {label:34} generated {m + unreach + no_anchor:5}  not measurable (CVE) {no_anchor:5}  "
          f"measurable {m:5}  0-1 hops {pct(sum(h <= 1 for h in hops), m):>6}  <=2 hops {pct(sum(h <= 2 for h in hops), m):>6}  "
          f"median {med}  unreachable {unreach:4} ({pct(unreach, a)} of anchored)  median, unreachable as >6: {med7}")

        line(f"  {src} documents", *rel(new[name], by_source=src))
    line("excluding ATT&CK parent/sibling ids", *rel(new[name], drop_family=True))
    h, _, _ = rel(new[name])
    print(f"  {'hop histogram (measurable)':34} {dict(sorted(Counter(h).items()))}")
